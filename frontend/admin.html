// BLOC 3 : Chargement et filtrage du Portefeuille (frontend/admin.js - Partie 1) const serverUrl = ""; // 1. Charger et filtrer dynamiquement les clients de cette agence précise async function chargerClients() { const tbody = document.getElementById('table-clients-body'); const banque = localStorage.getItem("cfg_banque") || "La Régionale"; const agence = localStorage.getItem("cfg_agence") || "Ambam"; try { const response = await fetch(`${serverUrl}/api/admin/clients?banque=${encodeURIComponent(banque)}&agence=${encodeURIComponent(agence)}`); const data = await response.json(); if (data.clients && data.clients.length > 0) { tbody.innerHTML = ""; data.clients.forEach(client => { const nombreDeMois = Math.floor(client.total_epargne / 20000); const fraisFormation = nombreDeMois * 5000; const epargneNette = client.total_epargne - fraisFormation; const droitCredit = epargneNette > 0 ? Math.floor((epargneNette / 0.15) * 0.85) : 0; tbody.innerHTML += ` #0${client.id} ${client.full_name}
${client.email} ${client.filiere} ${client.total_epargne.toLocaleString('fr-FR')} CFA -${fraisFormation.toLocaleString('fr-FR')} CFA ${droitCredit.toLocaleString('fr-FR')} CFA `; }); } else { tbody.innerHTML = 'Aucun agriculteur enregistré pour l\'agence de ' + agence + '.'; } } catch (error) { tbody.innerHTML = '⚠️ Erreur de connexion avec la base de données.'; } } // BLOC 4 : Inscription sécurisée multi-agences (frontend/admin.js - Partie 2) // 2. L'agriculteur a été inscrit pour l'agence courante."); nameInput.value = ""; emailInput.value = ""; garantieDetailsInput.value = ""; chargerClients(); } else { alert(result.detail || "❌ Échec de la validation par le serveur."); } } catch (error) { alert("❌ Erreur de connexion avec le serveur."); } finally { boutonSoumission.disabled = false; boutonSoumission.innerText = "Créer le compte d'accompagnement"; } } // BLOC 5 : Évaluation et extraction Excel (frontend/admin.js - Partie 3) // 3.
